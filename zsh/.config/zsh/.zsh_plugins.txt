#!/bin/zsh
#
# .zsh_plugins.txt - antidote plugins file
#

# Completion definitions: must be on fpath before compinit
zsh-users/zsh-completions kind:fpath path:src         # Extra community-maintained completion definitions
ohmyzsh/ohmyzsh kind:fpath path:plugins/extract       # _extract uses a passive #compdef, only found by compinit's fpath scan

# Base layer
mattmc3/z1 post:compinit                              # Options, history, keys, completion styles; compinit runs right after

# Completion UI (after compinit, before widget wrappers)
aloxaf/fzf-tab                                        # Replace zsh's default tab menu with an fzf-powered picker

# Utilities
zshzoo/macos conditional:is-macos                     # macOS-specific helpers (ofd, cdf, ...) loaded only on macOS
ohmyzsh/ohmyzsh path:plugins/extract                  # Universal `extract` command for any archive format
ohmyzsh/ohmyzsh path:plugins/git                      # Git aliases and helper functions (ga, gst, glog, …)

# Fish-like features (autosuggestions, then the highlighter last)
zsh-users/zsh-autosuggestions                         # Ghost-text suggestions from history/completion as you type
zdharma-continuum/fast-syntax-highlighting            # Real-time command syntax highlighting as you type
